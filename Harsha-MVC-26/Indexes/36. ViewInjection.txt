1. ViewInjection: We can use the service injection in views also.

home.cshtml
-----------

@using ServiceContracts

<h3 class="text-green">Injecting Service from Views</h3>

<div class="box">
    <h1>Cities</h1>
    @{
        List<string> vs = citiesServiceInView.GetCities();
    }
    <ul class="list">
        @foreach(var item in vs)
        {
            <li>@item</li>
        }
    </ul>
</div>
