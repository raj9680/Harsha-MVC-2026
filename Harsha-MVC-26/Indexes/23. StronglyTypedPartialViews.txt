1. StronglyTypedPartialViews: Strongly Typed Partial View is a partial view that is bound 
to a specified model class.

So. it gets all the benefits of a strongly typed view.


Index.cshtml
-------------

<div class="box">
    @*Way One: Returns the content to the parent view*@
    <partial name="_ListPartialView"></partial>

    @{
        var myViewData = new ViewDataDictionary(ViewData);
        myViewData["ListTitle"] = "Countries";
        myViewData["ListItems"] = new List<string>()
        {
        "USA",
        "Canada",
        "Japan",
        "India",
        "Germany"
        };
    }

    @*Invoking Partial View with Vide Data*@
    <partial name="_ListPartialView" view-data="myViewData"></partial>

    <h3>ListTitle in View: @ViewData["ListTitle"]</h3>
</div>





_ListPartialView.cshtml
------------------------

<div class="list-container">
    <h3>@ViewBag.ListTitle</h3>
    @{
        ViewBag.ListTitle = "Updated";
    }
    <ul class="list">
        @foreach (var item in ViewBag.ListItems)
        {
            <li>@item</li>
        }
    </ul>
</div>
